def step(state, action):
    s = state.clone()
    if action == "pickup(brass_key)":
        s.has_key, s.key_on_floor = True, False
    elif action == "activate(iron_door)":
        # Gate rule: door only opens when key is held
        if s.has_key:
            s.door_open = not s.door_open
    elif action == "toggle(redstone_lamp)":
        s.lamp_on = not s.lamp_on
    elif action == "camera.pan(away)":
        s.camera_occluded = not s.camera_occluded
    s.tick += 1
    return s, render(s)
